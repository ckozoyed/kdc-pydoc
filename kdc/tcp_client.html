 10class TcpClient:
 11    """
 12    TCP client for communicating with a KDC server.
 13
 14    TcpClient provides reliable, connection-oriented
 15    communication between Python applications and
 16    KDC servers.
 17
 18    The client supports:
 19        - Establishing and closing TCP connections.
 20        - Sending Agents asynchronously.
 21        - Sending Agents synchronously and waiting
 22          for their final Reply.
 23        - Receiving intermediate Replies and Agents
 24          during synchronous communication.
 25        - Notifying registered client event handlers.
 26        - Optional compression, encryption, and CRC
 27          through AgentSerializer.
 28
 29    Configuration is supplied through TcpClientParameters,
 30    normally constructed using TcpClientParametersBuilder.
 31
 32    Example:
 33        Create a TCP client and send an Agent::
 34
 35            parameters = (
 36                TcpClientParametersBuilder()
 37                .with_remote_address("localhost", 8001)
 38                .build()
 39            )
 40
 41            client = TcpClient(parameters)
 42
 43            try:
 44                client.connect()
 45
 46                agent = HelloAgent()
 47                reply = client.send_sync(agent)
 48
 49                print(reply.success)
 50
 51            finally:
 52                client.close()
 53
 54    Note:
 55        send_sync() blocks until the final matching Reply
 56        is received or an exception occurs.
 57
 58        send_async() sends an Agent without waiting for
 59        a Reply.
 60
 61        The current implementation does not provide a
 62        dedicated background receive thread.
 63    """
 64
 65    def __init__(self, params: TcpClientParameters):
 66        """
 67        Initialize a TCP client.
 68
 69        Creates the AgentSerializer and copies the
 70        configured event handlers.
 71
 72        The network connection is not established
 73        by the constructor.
 74
 75        Args:
 76            params (TcpClientParameters):
 77                TCP client configuration.
 78
 79        Raises:
 80            TypeError:
 81                If params is not TcpClientParameters.
 82        """
 83        if not isinstance(params, TcpClientParameters):
 84            raise TypeError("params must be TcpClientParameters")
 85
 86        self.params = params
 87        self.serializer = AgentSerializer(params)
 88
 89        self._socket = None
 90        self._connected = False
 91        self.event_handlers = list(params.get_event_handlers())

 97    def connect(self):
 98        """
 99        Establish a TCP connection to the configured server.
100
101        Creates a TCP socket and configures the connection
102        timeout, TCP_NODELAY, and TCP keep-alive settings.
103
104        After connecting, applies the configured read
105        timeout and notifies registered event handlers.
106
107        Raises:
108            RuntimeError:
109                If the client is already connected.
110            OSError:
111                If the socket cannot connect or configure
112                its network options.
113
114        Note:
115            If connection establishment fails, the socket
116            is closed and the exception is propagated.
117        """
118        if self._connected:
119            raise RuntimeError("TcpClient is already connected")
120
121        host, port = self.params.get_remote_address()
122
123        sock = socket.socket(
124            socket.AF_INET,
125            socket.SOCK_STREAM
126        )
127
128        try:
129            #
130            # Connect timeout
131            #
132            sock.settimeout(
133                self.params.get_connect_timeout_millis() / 1000.0
134            )
135
136            sock.setsockopt(
137                socket.IPPROTO_TCP,
138                socket.TCP_NODELAY,
139                int(self.params.is_tcp_no_delay())
140            )
141
142            sock.setsockopt(
143                socket.SOL_SOCKET,
144                socket.SO_KEEPALIVE,
145                int(self.params.is_keep_alive())
146            )
147
148            sock.connect((host, port))
149
150            #
151            # Switch to read timeout
152            #
153            sock.settimeout(
154                self.params.get_read_timeout_millis() / 1000.0
155            )
156
157            self._socket = sock
158            self._connected = True
159
160            for handler in self.event_handlers:
161                handler.on_connection_established(self)
162
163        except Exception:
164            sock.close()
165            self._socket = None
166            self._connected = False
167            raise

173    def send_async(self, agent: Agent):
174        """
175        Send an Agent asynchronously.
176
177        Marks the Agent as asynchronous, assigns the
178        SEND_ASYNC latency context, and transmits it
179        to the connected server.
180
181        Registered event handlers are notified after
182        the Agent has been sent.
183
184        Args:
185            agent (Agent):
186                Agent to transmit.
187
188        Raises:
189            RuntimeError:
190                If the client is not connected.
191            OSError:
192                If transmission fails.
193
194        Note:
195            This method does not wait for a Reply.
196            It does not create a background receiver.
197        """
198        agent.synchronous = False
199        agent.set_latency_context(LatencyContext.SEND_ASYNC)
200        self._send(agent)
201
202        for handler in self.event_handlers:
203            handler.on_sending_async_agent(agent)

205    def send_sync(self, agent: Agent):
206        """
207        Send an Agent and wait for its final Reply.
208
209        Marks the Agent as synchronous, assigns the
210        SEND_SYNC latency context, and transmits it.
211
212        After sending, the client receives messages
213        until a Reply matching the Agent UUID and
214        marked as the final Reply is received.
215
216        Intermediate Replies are delivered to registered
217        event handlers.
218
219        Incoming Agents are also delivered to registered
220        event handlers while waiting.
221
222        Args:
223            agent (Agent):
224                Agent to transmit.
225
226        Returns:
227            Reply:
228                Final Reply matching the transmitted Agent.
229
230        Raises:
231            RuntimeError:
232                If the client is not connected.
233            OSError:
234                If a network operation fails.
235            ConnectionError:
236                If the connection closes while receiving.
237            ValueError:
238                If an invalid packet length is received.
239            TypeError:
240                If a received object is neither Agent
241                nor Reply.
242
243        Note:
244            This method blocks while waiting for a
245            final matching Reply.
246
247            Socket reads are subject to the configured
248            read timeout. There is no separate overall
249            synchronous-operation timeout in this method.
250        """
251        agent.synchronous = True
252        agent.set_latency_context(LatencyContext.SEND_SYNC)
253        self._send(agent)
254
255        for handler in self.event_handlers:
256            handler.on_sending_sync_agent(agent)
257
258        while True:
259
260            obj = self._receive()
261
262            if isinstance(obj, Reply):
263
264                for handler in self.event_handlers:
265                    handler.on_received_reply(obj)
266
267                if obj.uuid == agent.uuid and obj.is_last_reply():
268                    return obj
269
270                # Unrelated/late Reply.
271                continue
272
273            if isinstance(obj, Agent):
274
275                for handler in self.event_handlers:
276                    handler.on_received_agent(obj)
277
278                continue
279
280            raise TypeError(
281                "TcpClient.send_sync: expected Agent or Reply"
282            )

284    def _send(self, agent):
285        """
286        Serialize and transmit an Agent over TCP.
287
288        Uses a four-byte signed big-endian length prefix
289        followed by the serialized Agent payload.
290
291        Args:
292            agent (Agent):
293                Agent to transmit.
294
295        Raises:
296            RuntimeError:
297                If the client is not connected.
298            OSError:
299                If socket transmission fails.
300        """
301        if not self._connected:
302            raise RuntimeError("TcpClient is not connected")
303
304        payload = self.serializer.serialize(agent)
305
306        length = len(payload).to_bytes(
307            4,
308            byteorder="big",
309            signed=True
310        )
311
312        try:
313            self._socket.sendall(length)
314            self._socket.sendall(payload)
315
316            self._socket.settimeout(
317                self.params.get_read_timeout_millis() / 1000.0
318            )
319
320        except OSError:
321            self._disconnect()
322            raise

328    def _receive(self):
329        """
330        Receive and deserialize one KDC message.
331
332        Reads the four-byte packet length, validates
333        the payload size, and receives the complete
334        serialized message.
335
336        Returns:
337            Agent | Reply:
338                Deserialized KDC message.
339
340        Raises:
341            ValueError:
342                If the packet length is invalid or
343                exceeds the configured maximum.
344            ConnectionError:
345                If the socket closes during reception.
346            OSError:
347                If a socket read fails.
348        """
349        length_bytes = self._read_exact(4)
350
351        length = int.from_bytes(
352            length_bytes,
353            byteorder="big",
354            signed=True
355        )
356
357        if length <= 0:
358            raise ValueError(
359                f"Invalid packet length: {length}"
360            )
361
362        max_size = self.params.get_max_received_buffer_size()
363
364        if length > max_size:
365            raise ValueError(
366                f"Packet length {length} exceeds "
367                f"maximum receive buffer size {max_size}"
368            )
369
370        payload = self._read_exact(length)
371
372        return self.serializer.deserialize(payload)

374    def _read_exact(self, size):
375        """
376        Read exactly the requested number of bytes.
377
378        Repeatedly receives socket data until the
379        requested number of bytes has been collected.
380
381        Args:
382            size (int):
383                Number of bytes to receive.
384
385        Returns:
386            bytes:
387                Received data.
388
389        Raises:
390            ConnectionError:
391                If the socket closes before all bytes
392                are received.
393            OSError:
394                If a socket read fails.
395        """
396        data = bytearray()
397
398        while len(data) < size:
399
400            try:
401                chunk = self._socket.recv(
402                    size - len(data)
403                )
404            except OSError:
405                self._disconnect()
406                raise
407
408            if not chunk:
409                self._disconnect()
410
411                raise ConnectionError(
412                    "Socket closed while reading"
413                )
414
415            data.extend(chunk)
416
417        return bytes(data)

TCP client for communicating with a KDC server.

TcpClient provides reliable, connection-oriented communication between Python applications and KDC servers.

The client supports: - Establishing and closing TCP connections. - Sending Agents asynchronously. - Sending Agents synchronously and waiting for their final Reply. - Receiving intermediate Replies and Agents during synchronous communication. - Notifying registered client event handlers. - Optional compression, encryption, and CRC through AgentSerializer.

Configuration is supplied through TcpClientParameters, normally constructed using TcpClientParametersBuilder.

Example: Create a TCP client and send an Agent::

    parameters = (
        TcpClientParametersBuilder()
        .with_remote_address("localhost", 8001)
        .build()
    )

    client = TcpClient(parameters)

    try:
        client.connect()

        agent = HelloAgent()
        reply = client.send_sync(agent)

        print(reply.success)

    finally:
        client.close()

Note: send_sync() blocks until the final matching Reply is received or an exception occurs.

send_async() sends an Agent without waiting for
a Reply.

The current implementation does not provide a
dedicated background receive thread.
