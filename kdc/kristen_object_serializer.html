 18    """
 19    Internal object serializer for KDC-Python.
 20
 21    Provides JSON serialization and deserialization,
 22    optional GZIP compression, and AES-GCM authenticated
 23    encryption.
 24
 25    Mirrors the functionality of
 26    com.kristensoft.kristen.objects.KristenObjectsSerializer
 27    in KDC-Java.
 28
 29    Serialization order:
 30        1. Convert object attributes to wire-compatible values.
 31        2. Encode the resulting dictionary as UTF-8 JSON.
 32        3. Optionally compress using GZIP.
 33        4. Optionally encrypt using AES-GCM.
 34
 35    Deserialization reverses these operations.
 36
 37    Wire field naming precedence:
 38        1. Explicit mappings defined by WIRE_FIELDS.
 39        2. Automatic snake_case to camelCase translation
 40           when case_translation is enabled.
 41        3. Original Python attribute names otherwise.
 42
 43    Supported value conversions include:
 44        - UUID values to and from strings.
 45        - Enum values to and from their member names.
 46        - Dictionaries, lists, and tuples.
 47        - Nested objects with instance attributes.
 48        - LatencyRecord reconstruction using type hints.
 49
 50    AES-GCM format:
 51        IV (12 bytes) + ciphertext + authentication tag (16 bytes).
 52
 53    Note:
 54        This class is intended for internal KDC use.
 55        Applications should normally use the higher-level
 56        KDC serialization and communication APIs.
 57    """

161    def serialize(self, obj: Any) -> bytes:
162        """
163        Serialize an object into wire-compatible bytes.
164
165        Converts object attributes into a dictionary,
166        applies wire field naming rules, and encodes
167        the result as UTF-8 JSON.
168
169        Optional compression and encryption are applied
170        after JSON encoding.
171
172        Args:
173            obj (Any):
174                Object to serialize.
175
176        Returns:
177            bytes:
178                Serialized object bytes.
179
180        Raises:
181            ValueError:
182                If obj is None.
183            RuntimeError:
184                If encryption is enabled without a key.
185        """
186        if obj is None:
187            raise ValueError("serialize: object is None")
188
189        mapping = self._get_wire_fields(type(obj)) or {}
190
191        wire = {}
192
193        for python_name, value in vars(obj).items():
194
195            if python_name in mapping:
196                wire_name = mapping[python_name]
197            elif self.case_translation:
198                wire_name = self._snake_to_camel(python_name)
199            else:
200                wire_name = python_name
201
202            wire[wire_name] = self._to_wire_value(value)
203
204        # 1. JSON
205        data = json.dumps(
206            wire,
207            separators=(",", ":")
208        ).encode("utf-8")
209
210        # 2. Compress
211        if self.compression:
212            data = self._compress(data)
213
214        # 3. Encrypt
215        if self.encrypted_mode:
216            data = self._encrypt(data)
217
218        return data

224    def deserialize(self, data: bytes, clazz):
225        """
226        Deserialize wire-compatible bytes into an object.
227
228        Optionally decrypts and decompresses the input,
229        decodes the UTF-8 JSON representation, and
230        reconstructs an instance of the specified class.
231
232        Field names are resolved using WIRE_FIELDS
233        mappings or automatic case translation.
234
235        Type hints are used when reconstructing supported
236        types such as UUID, Enum, and LatencyRecord.
237
238        Args:
239            data (bytes):
240                Serialized object bytes.
241            clazz (type):
242                Class to instantiate.
243
244        Returns:
245            object:
246                Reconstructed instance of clazz.
247
248        Raises:
249            ValueError:
250                If data is None.
251            RuntimeError:
252                If encryption is enabled without a key.
253
254        Note:
255            clazz must support construction without
256            required positional arguments.
257        """
258        if data is None:
259            raise ValueError("deserialize: data is None")
260
261        # 1. Decrypt
262        if self.encrypted_mode:
263            data = self._decrypt(data)
264
265        # 2. Decompress
266        if self.compression:
267            data = self._decompress(data)
268
269        # 3. JSON -> dict
270        data_dict = json.loads(data.decode("utf-8"))
271
272        # 4. Create object
273        obj = clazz()
274
275        type_hints = get_type_hints(clazz)
276
277        # 5. Reverse wire mapping
278        mapping = self._get_wire_fields(clazz)
279        reverse_mapping = {
280            wire: py for py, wire in mapping.items()
281        }
282
283        # 6. Populate object
284        for wire_name, value in data_dict.items():
285
286            python_name = (
287                reverse_mapping.get(wire_name) or (
288                    self._camel_to_snake(wire_name)
289                    if self.case_translation
290                    else wire_name
291                )
292            )
293
294            expected_type = type_hints.get(python_name)
295
296            value = self._from_wire_value(
297                value,
298                expected_type
299            )
300
301            setattr(obj, python_name, value)
302
303        return obj

309    def _compress(self, data: bytes) -> bytes:
310        """
311        Compress bytes using GZIP.
312
313        Args:
314            data (bytes):
315                Uncompressed input bytes.

317        Returns:
318            bytes:
319                GZIP-compressed bytes.

322

329        """
330        Decompress GZIP-encoded bytes.

332        Args:
333            data (bytes):
334                GZIP-compressed bytes.
335
336        Returns:
337            bytes:
338                Decompressed bytes.
339        """
340        with gzip.GzipFile(
341            fileobj=BytesIO(data),

344            return gz.read()

350    def _encrypt(self, plaintext: bytes) -> bytes:
351        """
352        Encrypt bytes using AES-GCM.
353
354        Generates a random 12-byte IV and appends
355        the 16-byte authentication tag.
356
357        Args:
358            plaintext (bytes):
359                Data to encrypt.
360
361        Returns:
362            bytes:
363                IV + ciphertext + authentication tag.
364
365        Raises:
366            RuntimeError:
367                If the encryption key is not configured.
368        """
369        if self.encryption_key is None:
370            raise RuntimeError("Encryption key not set")
371
372        iv = get_random_bytes(self.GCM_IV_LENGTH)
373
374        cipher = AES.new(
375            self.encryption_key,
376            AES.MODE_GCM,
377            nonce=iv
378        )
379
380        ciphertext, tag = cipher.encrypt_and_digest(
381            plaintext
382        )
383
384        return iv + ciphertext + tag
385
386    def _decrypt(self, data: bytes) -> bytes:
387        """
388        Decrypt and authenticate AES-GCM data.
389
390        Expects the wire format:
391            IV + ciphertext + authentication tag.
392
393        Args:
394            data (bytes):
395                AES-GCM encrypted bytes.
396
397        Returns:
398            bytes:
399                Decrypted plaintext.
400
401        Raises:
402            RuntimeError:
403                If the encryption key is not configured.
404            ValueError:
405                If authentication fails.
406        """
407        if self.encryption_key is None:
408            raise RuntimeError("Encryption key not set")
409
410        iv = data[:self.GCM_IV_LENGTH]
411        tag = data[-self.GCM_TAG_LENGTH:]
412        ciphertext = data[
413            self.GCM_IV_LENGTH:-self.GCM_TAG_LENGTH
414        ]
415
416        cipher = AES.new(
417            self.encryption_key,
418            AES.MODE_GCM,
419            nonce=iv
420        )
421
422        return cipher.decrypt_and_verify(
423            ciphertext,
424            tag
425        )

431    @staticmethod
432    def _get_wire_fields(clazz):
433        """
434        Collect wire field mappings from a class hierarchy.
435
436        Base class mappings are applied first, allowing
437        derived classes to override inherited mappings.
438
439        Args:
440            clazz (type):
441                Class whose mappings are requested.
442
443        Returns:
444            dict:
445                Python attribute names mapped to wire names.
446        """
447        mapping = {}
448
449        for c in reversed(clazz.__mro__):
450            mapping.update(
451                getattr(c, "WIRE_FIELDS", {})
452            )
453
454        return mapping

456    @staticmethod
457    def _snake_to_camel(name: str) -> str:
458        """
459        Convert a snake_case name to camelCase.
460
461        Args:
462            name (str):
463                Python attribute name.
464
465        Returns:
466            str:
467                CamelCase-compatible wire field name.
468        """
469        if "_" not in name:
470            return name
471
472        parts = name.split("_")
473
474        return parts[0] + "".join(
475            part.capitalize()
476            for part in parts[1:]
477        )
478
479    @staticmethod
480    def _camel_to_snake(name: str) -> str:
481        """
482        Convert a camelCase name to snake_case.
483
484        Args:
485            name (str):
486                Wire field name.
487
488        Returns:
489            str:
490                Snake_case Python attribute name.
491        """
492        _CAMEL_CASE_PATTERN = re.compile(
493            r'(?<!^)(?=[A-Z])'
494        )
495
496        return _CAMEL_CASE_PATTERN.sub(
497            "_",
498            name
499        ).lower()

505    def _from_wire_value(self, value, expected_type):
506        """
507        Convert a wire value into its expected Python type.
508
509        Uses type hints to reconstruct supported values,
510        including UUID, Enum, and LatencyRecord.
511
512        Args:
513            value:
514                Value decoded from JSON.
515            expected_type:
516                Expected Python type, if available.
517
518        Returns:
519            object:
520                Converted value or the original value
521                when no conversion is applicable.
522        """
523        if expected_type is None:
524            return value
525
526        # Handle X | None
527        origin = get_origin(expected_type)
528
529        if origin is types.UnionType:
530
531            for candidate in get_args(expected_type):
532
533                if candidate is type(None):
534                    continue
535
536                return self._from_wire_value(
537                    value,
538                    candidate
539                )
540
541            return value
542
543        if expected_type is UUID and isinstance(value, str):
544            return UUID(value)
545
546        if (
547            isinstance(expected_type, type)
548            and issubclass(expected_type, Enum)
549            and isinstance(value, str)
550        ):
551            return expected_type[value]
552
553        if (
554            expected_type is LatencyRecord
555            and isinstance(value, dict)
556        ):
557            context = LatencyContext[
558                value["latencyContext"]
559            ]
560
561            record = LatencyRecord(context)
562
563            record.start_time_line = list(
564                value["startTimeLine"]
565            )
566
567            record.stop_time_line = list(
568                value["stopTimeLine"]
569            )
570
571            return record
572
573        return value
574
575    def _to_wire_value(self, value):
576        """
577        Convert a Python value into a JSON-compatible value.
578
579        Handles UUID, Enum, dictionaries, sequences,
580        and nested objects.
581
582        Nested object fields are converted using
583        WIRE_FIELDS mappings or snake_case to camelCase
584        translation.
585
586        Args:
587            value:
588                Python value to convert.
589
590        Returns:
591            object:
592                JSON-compatible representation.
593        """
594        if isinstance(value, UUID):
595            return str(value)
596
597        if isinstance(value, Enum):
598            return value.name
599
600        if isinstance(value, dict):
601            return {
602                key: self._to_wire_value(item)
603                for key, item in value.items()
604            }
605
606        if isinstance(value, (list, tuple)):
607            return [
608                self._to_wire_value(item)
609                for item in value
610            ]
611
612        if hasattr(value, "__dict__"):
613
614            mapping = (
615                self._get_wire_fields(type(value)) or {}
616            )
617
618            wire = {}
619
620            for python_name, item in vars(value).items():
621
622                if python_name in mapping:
623                    wire_name = mapping[python_name]
624                else:
625                    wire_name = self._snake_to_camel(
626                        python_name
627                    )
628
629                wire[wire_name] = self._to_wire_value(
630                    item
631                )
632
633            return wire
634
635        return value

512        including UUID, Enum, and LatencyRecord.

518                Expected Python type, if available.
519

522                Converted value or the original value
523                when no conversion is applicable.
524        """

526            return value
527
528        # Handle X | None

531        if origin is types.UnionType:
532

534

539                    value,

541                )

577    def _to_wire_value(self, value):
578        """
579        Convert a Python value into a JSON-compatible value.
580
581        Handles UUID, Enum, dictionaries, sequences,
582        and nested objects.
583
584        Nested object fields are converted using
585        WIRE_FIELDS mappings or snake_case to camelCase
586        translation.
587
588        Args:
589            value:
590                Python value to convert.
591
592        Returns:
593            object:
594                JSON-compatible representation.
595        """
596        if isinstance(value, UUID):
597            return str(value)
598
599        if isinstance(value, Enum):
600            return value.name
601
602        if isinstance(value, dict):
603            return {
604                key: self._to_wire_value(item)
605                for key, item in value.items()
606            }
607
608        if isinstance(value, (list, tuple)):
609            return [
610                self._to_wire_value(item)
611                for item in value
612            ]
613
614        if hasattr(value, "__dict__"):
615
616            mapping = (
617                self._get_wire_fields(type(value)) or {}
618            )
619
620            wire = {}
621
622            for python_name, item in vars(value).items():
623
624                if python_name in mapping:
625                    wire_name = mapping[python_name]
626                else:
627                    wire_name = self._snake_to_camel(
628                        python_name
629                    )
630
631                wire[wire_name] = self._to_wire_value(
632                    item
633                )
634
635            return wire
636
637        return value

Internal object serializer for KDC-Python.

Provides JSON serialization and deserialization, optional GZIP compression, and AES-GCM authenticated encryption.

Mirrors the functionality of com.kristensoft.kristen.objects.KristenObjectsSerializer in KDC-Java.

Serialization order: 1. Convert object attributes to wire-compatible values. 2. Encode the resulting dictionary as UTF-8 JSON. 3. Optionally compress using GZIP. 4. Optionally encrypt using AES-GCM.

Deserialization reverses these operations.

Wire field naming precedence: 1. Explicit mappings defined by WIRE_FIELDS. 2. Automatic snake_case to camelCase translation when case_translation is enabled. 3. Original Python attribute names otherwise.

Supported value conversions include: - UUID values to and from strings. - Enum values to and from their member names. - Dictionaries, lists, and tuples. - Nested objects with instance attributes. - LatencyRecord reconstruction using type hints.

AES-GCM format: IV (12 bytes) + ciphertext + authentication tag (16 bytes).

Note: This class is intended for internal KDC use. Applications should normally use the higher-level KDC serialization and communication APIs.
